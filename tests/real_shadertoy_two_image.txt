#define M_PI 3.14159265358979323846264338327950288

// Function to rotate UV coordinates around a given point (rotCenter)
vec2 rotateUV(vec2 uv, vec2 center, mat2 rotationMatrix) {
    uv -= center; // Move to origin for rotation
    uv = rotationMatrix * uv; // Rotate UV coordinates
    uv += center; // Move back to original position
    return uv;
}

// Easing Functions

float easeInOutQuint(float x) {
    if (x < 0.5) {
        return 16.0 * x * x * x * x * x;
    } else {
        return 1.0 - pow(-2.0 * x + 2.0, 5.0) * 0.5;
    }
}

float easeInOutBack(float x) {
float c1 = 1.70158;
float c2 = c1 * 1.525;

return x < 0.5
  ? (pow(2.0 * x, 2.0) * ((c2 + 1.0) * 2. * x - c2)) / 2.
  : (pow(2. * x - 2., 2.) * ((c2 + 1.) * (x * 2. - 2.) + c2) + 2.) / 2.;
}

float easeInOutElastic(float x) {
    const float c5 = (2.0 * 3.14159) / 4.5;

    if (x == 0.0) {
        return 0.0;
    } else if (x == 1.0) {
        return 1.0;
    } else if (x < 0.5) {
        return -0.5 * pow(2.0, 20.0 * x - 10.0) * sin((20.0 * x - 11.125) * c5);
    } else {
        return 0.5 * pow(2.0, -20.0 * x + 10.0) * sin((20.0 * x - 11.125) * c5) + 1.0;
    }
}

mat2 getRot(float angle) {
    return mat2(cos(angle), -sin(angle), sin(angle), cos(angle));
}

float smooth_transition_oscillate(float t, float duration, float start, float end) {
    float v1 = mod(t, duration);
    float v2 = mod(t, duration * 2.0);
    float newT = v2 > v1 ? duration - v1 : v1;

    // The transition points also need to be cyclic, using newT for checking
    float transitionStart = start; // duration / 2.0 - 0.5; // Transition starts at 2
    float transitionEnd = end; // duration / 2.0 + 0.5;   // Transition ends at 3

    if (newT < transitionStart) {
        return 0.; // Before the transition starts
    } else if (newT >= transitionStart && newT < transitionEnd) {
        // Linear interpolation between 0 and 1 for the transition period
        return (newT - transitionStart) / (transitionEnd - transitionStart);
    } else {
        return 1.; // After the transition ends
    }
}

void mainImage(out vec4 fragColor, in vec2 fragCoord) {
    // Aspect ratio correction
    float aspectRatio = iResolution.x / iResolution.y;
    vec2 uv = fragCoord / iResolution.xy;
    uv.x *= aspectRatio; // Adjust the x coordinate by aspect ratio

    // Calculate the distance from the current point to the center
    // Adjusted center.x for aspect ratio
    float dist = length((fragCoord.xy / iResolution.y) - vec2(aspectRatio * 0.5, 0.5));

    // Define radii for the three circles
    float outerRadius = 0.41;
    float middleRadius = 0.31;
    float innerRadius = 0.;

    // Calculate the ring's thickness
    float thickness = 0.05; // Thickness based on the normalized units

    // Check if we're within the ring's thickness for each circle
    bool inOuterRing = abs(dist - outerRadius) < thickness * 0.7;
    bool inMiddleRing = abs(dist - middleRadius) < thickness * 1.3;
    bool inInnerRing = abs(dist - innerRadius) < thickness * 5.0;

    // Calculate angle for rotation
    float angle = iTime; // Rotate based on time
    
    // oscillators
    float transitionDuration = 5.0; // Total duration for a full cycle (up and back down)
    float period = 10.0; // Total duration for a full cycle (up and back down)

    // Calculating the position for bg texture
    vec2 texRes = vec2(iChannelResolution[0].xy); // Assuming both 0, 1 textures have same resolutions
    float scaleToFit = iResolution.y / texRes.y;
    float xOffset = iResolution.x * 0.5 - (texRes.x * scaleToFit) * 0.5;
    vec2 uvTex = (fragCoord - vec2(xOffset, 0.0)) / scaleToFit;
    
    // Calculate screen center adjusted for aspect ratio and scaling
    vec2 screenCenter = vec2((iResolution.x) * 0.5, iResolution.y * 0.5);
    
    // for rotating
    vec2 rotCenter = vec2(0.5, 0.5);
    
    // Calculate factors with staggered start times and special easing
    float mix1 = smooth_transition_oscillate(iTime, transitionDuration, transitionDuration * 0.2, transitionDuration * 0.8);
    float mix2 = smooth_transition_oscillate(iTime, transitionDuration, transitionDuration * 0.3, transitionDuration * 0.7);
    float mix3 = smooth_transition_oscillate(iTime, transitionDuration, transitionDuration * 0.4, transitionDuration * 0.6);

    // Adding more behavior
    mix1 = easeInOutElastic(mix1);
    mix2 = easeInOutElastic(mix2);
    mix3 = easeInOutQuint(mix3);

    // Apply rotation only within the circles
    vec2 rot1Uv;
    vec2 rot2Uv;
    vec2 rot3Uv;
    if (inOuterRing || inMiddleRing || inInnerRing) {
        // Convert screen coordinates to normalized UV coordinates for rotation
        vec2 uvTexCircle = (fragCoord - screenCenter) / iResolution.y + rotCenter;
        rot1Uv = rotateUV(uvTexCircle, rotCenter, getRot(mix1 * 2. * M_PI));
        rot2Uv = rotateUV(uvTexCircle, rotCenter, getRot(mix2 * 2. * M_PI * -1.));
        rot3Uv = rotateUV(uvTexCircle, rotCenter, getRot(mix3 * 2. * M_PI));
    }
    
    float mix4Duration = 0.07;
    float mix4 = smooth_transition_oscillate(iTime, transitionDuration, transitionDuration / 2. - mix4Duration, transitionDuration / 2. + mix4Duration);

    // Output to screen
    if (inOuterRing) {
        fragColor = mix(texture(iChannel0, rot3Uv), texture(iChannel1, rot3Uv), mix3);
    } else if (inMiddleRing) {
        fragColor = mix(texture(iChannel0, rot2Uv), texture(iChannel1, rot2Uv), mix2);
    } else if (inInnerRing) {
        fragColor = mix(texture(iChannel0, rot1Uv), texture(iChannel1, rot1Uv), mix1);
    } else {
        // Textures for the background
        uvTex /= texRes;
        vec4 tex1bgColor = texture(iChannel0, uvTex);
        vec4 tex2bgColor = texture(iChannel1, uvTex);
        fragColor = mix(tex1bgColor, tex2bgColor, mix4); // Mix the two textures for the background
    }
}
